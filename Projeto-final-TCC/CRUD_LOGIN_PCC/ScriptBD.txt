-- ============================================================
-- 7. PRE_MATRICULA
-- O ALUNO realiza a pre-matricula.
-- A Secretaria analisa posteriormente.
-- ============================================================

CREATE TABLE pre_matricula (
    id_pre_matricula INT AUTO_INCREMENT PRIMARY KEY,

    id_usuario INT NOT NULL,

    id_curso INT NOT NULL,

    status ENUM(
        'Em Analise',
        'Pendente de Documentacao',
        'Aprovado',
        'Indeferido'
    ) DEFAULT 'Em Analise',

    observacao TEXT,

    data_inscricao TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    data_analise DATETIME,

    analisado_por INT,

    CONSTRAINT fk_pre_matricula_aluno
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
        ON DELETE CASCADE,

    CONSTRAINT fk_pre_matricula_curso
        FOREIGN KEY (id_curso)
        REFERENCES cursos(id_curso)
        ON DELETE RESTRICT,

    CONSTRAINT fk_pre_matricula_analisador
        FOREIGN KEY (analisado_por)
        REFERENCES usuarios(id_usuario)
        ON DELETE SET NULL
) ENGINE=InnoDB;

-- ============================================================
-- 9. MATRICULAS

        'Ativo',
        'Trancado',
        'Cancelado',

    CONSTRAINT fk_matricula_aluno
        FOREIGN KEY (id_usuario)
        REFERENCES usuarios(id_usuario)
        ON DELETE RESTRICT,

    CONSTRAINT fk_matricula_pre_matricula
        FOREIGN KEY (id_pre_matricula)

        ON DELETE SET NULL,

    CONSTRAINT uq_aluno_periodo
        UNIQUE (
            id_usuario,

        ),

    CONSTRAINT uq_pre_matricula
        UNIQUE (id_pre_matricula)

-- ============================================================
-- 14. TRIGGER - VALIDAR PRE-MATRICULA
-- Somente usuarios com perfil ALUNO podem realizar
-- uma pre-matricula.
-- ============================================================

DELIMITER $$

CREATE TRIGGER trg_validar_pre_matricula
BEFORE INSERT ON pre_matricula
FOR EACH ROW
BEGIN

    DECLARE v_perfil INT;

    SELECT id_perfil
    INTO v_perfil
    FROM usuarios
    WHERE id_usuario = NEW.id_usuario;

    IF v_perfil IS NULL THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
            'Usuario nao encontrado';

    END IF;

    IF v_perfil <> 3 THEN

        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
            'A pre-matricula deve ser realizada por um usuario com perfil Aluno';

    END IF;

END$$

DELIMITER ;


-- ============================================================
-- 15. TRIGGER - VALIDAR MATRICULA
-- ============================================================

DELIMITER $$

CREATE TRIGGER trg_validar_matricula
BEFORE INSERT ON matriculas
FOR EACH ROW
BEGIN

    DECLARE v_status VARCHAR(50);

    DECLARE v_usuario_pre INT;

    DECLARE v_perfil INT;


    -- ========================================================
    -- PRIMEIRA MATRICULA
    -- ========================================================

    IF NEW.tipo_matricula = 'Primeira Matricula' THEN


        -- Deve possuir pre-matricula
        IF NEW.id_pre_matricula IS NULL THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'Primeira matricula deve possuir uma pre-matricula';

        END IF;


        -- Busca a pre-matricula
        SELECT
            status,
            id_usuario
        INTO
            v_status,
            v_usuario_pre
        FROM pre_matricula
        WHERE id_pre_matricula = NEW.id_pre_matricula;


        -- Pre-matricula nao encontrada
        IF v_usuario_pre IS NULL THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'Pre-matricula nao encontrada';

        END IF;


        -- Pre-matricula precisa estar aprovada
        IF v_status <> 'Aprovado' THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'Pre-matricula ainda nao foi aprovada pela secretaria';

        END IF;


        -- Verifica se o usuario da matricula e ALUNO
        SELECT id_perfil
        INTO v_perfil
        FROM usuarios
        WHERE id_usuario = NEW.id_usuario;


        IF v_perfil IS NULL THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'Usuario nao encontrado';

        END IF;


        IF v_perfil <> 3 THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'O usuario da matricula deve possuir o perfil Aluno';

        END IF;


        -- O usuario da matricula deve ser
        -- o mesmo que realizou a pre-matricula
        IF NEW.id_usuario <> v_usuario_pre THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'O aluno nao corresponde ao usuario da pre-matricula';

        END IF;


    -- ========================================================
    -- REMATRICULA
    -- ========================================================

    ELSEIF NEW.tipo_matricula = 'Rematricula' THEN


        -- Rematricula nao utiliza pre-matricula
        IF NEW.id_pre_matricula IS NOT NULL THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'Rematricula nao deve possuir pre-matricula';

        END IF;


        -- O usuario precisa existir
        SELECT id_perfil
        INTO v_perfil
        FROM usuarios
        WHERE id_usuario = NEW.id_usuario;


        IF v_perfil IS NULL THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'Usuario nao encontrado';

        END IF;


        -- O usuario precisa ser ALUNO
        IF v_perfil <> 3 THEN

            SIGNAL SQLSTATE '45000'
            SET MESSAGE_TEXT =
                'O usuario da rematricula deve possuir o perfil Aluno';

        END IF;

    END IF;

END$$

DELIMITER ;
